cmake_minimum_required(VERSION 3.16)
project(BDS5GPositioning VERSION 0.2.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)
option(BDS5G_BUILD_5G "Build 5G processing and replay" ON)
option(BDS5G_BUILD_RADIO "Build live USRP receiver and anchor (requires UHD)" ON)
option(BDS5G_BUILD_GNSS "Build BeiDou B1C offline acquisition, tracking, navigation and SPP" OFF)
option(BDS5G_BUILD_FUSION "Build reserved fusion module" OFF)
option(BDS5G_BUILD_TESTS "Build offline tests" ON)
set(BDS5G_MAIN_CONFIG "${CMAKE_CURRENT_SOURCE_DIR}/config/config.f" CACHE FILEPATH "Default main configuration")
if(BDS5G_BUILD_TESTS)
  enable_testing()
endif()
if(BDS5G_BUILD_5G)
  add_subdirectory(src/5G)
endif()
if(BDS5G_BUILD_GNSS OR BDS5G_BUILD_FUSION)
  add_subdirectory(src/fusion)
endif()
if(BDS5G_BUILD_GNSS)
  add_subdirectory(src/GNSS)
endif()
find_package(Python3 3.9 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_custom_target(nr-monitor-ui
    COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/ui/server.py" --config "${BDS5G_MAIN_CONFIG}"
    WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}" USES_TERMINAL)
  if(BDS5G_BUILD_TESTS)
    add_test(NAME monitor_service
      COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${PROJECT_SOURCE_DIR}/ui/tests" -v)
  endif()
endif()

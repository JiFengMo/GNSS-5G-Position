find_package(PkgConfig REQUIRED)
pkg_check_modules(FFTW3 REQUIRED IMPORTED_TARGET fftw3)
enable_language(CXX)
find_package(Threads REQUIRED)
add_library(bds_b1c STATIC bds_b1c/b1c_code.c bds_b1c/b1c_acquisition.c bds_b1c/b1c_parallel.cpp bds_b1c/b1c_tracking.c bds_b1c/b1c_navigation.c bds_b1c/b1c_position.c)
target_compile_features(bds_b1c PRIVATE cxx_std_17)
target_include_directories(bds_b1c PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(bds_b1c PUBLIC bds5g_observation PkgConfig::FFTW3 Threads::Threads)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(bds_b1c PRIVATE -Wall -Wextra)
endif()
if(NOT MSVC)
  target_link_libraries(bds_b1c PUBLIC m)
endif()

target_compile_definitions(bds-b1c-track PRIVATE _FILE_OFFSET_BITS=64)

find_package(PkgConfig REQUIRED)
pkg_check_modules(FFTW3 REQUIRED IMPORTED_TARGET fftw3)
add_library(bds_b1c STATIC bds_b1c/b1c_code.c bds_b1c/b1c_acquisition.c bds_b1c/b1c_tracking.c bds_b1c/b1c_navigation.c bds_b1c/b1c_position.c)
target_include_directories(bds_b1c PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(bds_b1c PUBLIC bds5g_observation PkgConfig::FFTW3)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(bds_b1c PRIVATE -Wall -Wextra)
endif()
if(NOT MSVC)
  target_link_libraries(bds_b1c PUBLIC m)
endif()
add_executable(bds-b1c-receiver app/bds_b1c_receiver.c)
target_link_libraries(bds-b1c-receiver PRIVATE bds_b1c)
target_compile_definitions(bds-b1c-receiver PRIVATE _FILE_OFFSET_BITS=64)
add_executable(bds-b1c-track app/bds_b1c_track.c)
target_link_libraries(bds-b1c-track PRIVATE bds_b1c)
add_executable(bds-b1c-nav app/bds_b1c_nav.c)
target_link_libraries(bds-b1c-nav PRIVATE bds_b1c)
add_executable(bds-b1c-position app/bds_b1c_position.c)
target_link_libraries(bds-b1c-position PRIVATE bds_b1c)
target_compile_definitions(bds-b1c-track PRIVATE _FILE_OFFSET_BITS=64)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(bds-b1c-track PRIVATE -Wall -Wextra -Wformat=2)
  target_compile_options(bds-b1c-nav PRIVATE -Wall -Wextra -Wformat=2)
  target_compile_options(bds-b1c-position PRIVATE -Wall -Wextra -Wformat=2)
endif()

  add_test(NAME b1c_code COMMAND test_b1c_code

find_package(Threads REQUIRED)
file(GLOB NR_POSITIONING_SOURCES CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/positioning/common/*.c"
  "${CMAKE_CURRENT_SOURCE_DIR}/positioning/ssb/*.c"
  "${CMAKE_CURRENT_SOURCE_DIR}/positioning/prs/*.c")
# Offline DSP has no UHD dependency.
list(REMOVE_ITEM NR_POSITIONING_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/positioning/common/nr_radio_if.c")
add_library(nr_positioning STATIC ${NR_POSITIONING_SOURCES})
target_include_directories(nr_positioning PUBLIC
  "${CMAKE_CURRENT_SOURCE_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}"
  "${CMAKE_CURRENT_SOURCE_DIR}/third_party/oai")
target_compile_definitions(nr_positioning PUBLIC _GNU_SOURCE NR_DEFAULT_MAIN_CONFIG="${BDS5G_MAIN_CONFIG}")
target_link_libraries(nr_positioning PUBLIC Threads::Threads)
if(NOT MSVC)
  target_compile_options(nr_positioning PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(nr_positioning PUBLIC m)
endif()
if(WIN32 AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(nr_positioning PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/include/nr_windows_compat.h")
endif()
add_executable(replay_toa_iq tools/replay_toa_iq.c)
target_link_libraries(replay_toa_iq PRIVATE nr_positioning)
add_executable(nr-config-check tools/nr_config_check.c)
target_link_libraries(nr-config-check PRIVATE nr_positioning)
if(BDS5G_BUILD_TESTS)
  foreach(name ssb_sync ssb_toa prs_toa solver provider_select config_router)
    add_executable(test_${name} tests/test_${name}.c)
    target_link_libraries(test_${name} PRIVATE nr_positioning)
    add_test(NAME nr_${name} COMMAND test_${name})
  endforeach()
  set_tests_properties(nr_config_router PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
endif()
if(BDS5G_BUILD_RADIO)
  enable_language(CXX)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(UHD REQUIRED IMPORTED_TARGET uhd)
  add_library(nr_radio STATIC radio/USRP/usrp_lib.cpp radio/LIME/lime_lib.cpp positioning/common/nr_radio_if.c)
  target_compile_features(nr_radio PRIVATE cxx_std_17)
  target_link_libraries(nr_radio PUBLIC nr_positioning PkgConfig::UHD)
  foreach(name nr-toa-uesoftmodem nr-ssb-anchor)
    add_executable(${name} app/${name}.c)
    target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/app")
    target_link_libraries(${name} PRIVATE nr_radio)
  endforeach()
endif()
